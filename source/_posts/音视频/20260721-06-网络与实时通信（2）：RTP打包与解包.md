# 问题：一个视频帧塞不进一个 RTP 包

网络层的 MTU（Maximum Transmission Unit）通常是 1500 字节。减去 IP 头（20）+ UDP 头（8）+ RTP 头（12），单个 RTP 包的可用负载只有约 **1460 字节**。

但 H.264 的一个关键帧（I 帧）动辄几十 KB，根本塞不进一个包。所以必须**分片**——把一帧拆成多个 RTP 包，接收端再拼回来。

H.264 定义了三种 RTP 打包模式：

| 模式 | NAL Unit 类型 | 说明 |
|------|---------------|------|
| **单一 NAL 单元** | 1~23（原始类型） | 小帧，一个 RTP 包恰好装一个 NALU |
| **FU-A（分片单元）** | 28 | 大帧拆成多片，最常用 |
| **STAP-A（聚合包）** | 24 | 多个小 NALU 打包到一个 RTP 包 |

FU-A 是使用最广泛的分片模式。一个 50KB 的 I 帧会被拆成约 35 个 FU-A 包，每个包带 1400 字节的视频数据。

# FU-A 分片结构

```
原始 NAL Unit（如一个 I 帧）:
┌──────┬──────────────────────────────────┐
│ NAL  │        NAL 负载数据              │
│Header│  (几千到几十 KB)                  │
└──────┴──────────────────────────────────┘

拆成 FU-A 包:
┌─────────────────────────────────────────┐
│ FU-A 包 1 (Start)                       │
│ ┌──────┬───────┬──────────────────┐    │
│ │  FU   │  FU   │                   │    │
│ │indicator│header│  分片数据 #1       │    │
│ └──────┴───────┴──────────────────┘    │
├─────────────────────────────────────────┤
│ FU-A 包 2..N-1 (Middle)                 │
│ ┌──────┬───────┬──────────────────┐    │
│ │  FU   │  FU   │                   │    │
│ │indicator│header│  分片数据 #2..#N-1  │    │
│ └──────┴───────┴──────────────────┘    │
├─────────────────────────────────────────┤
│ FU-A 包 N (End)                         │
│ ┌──────┬───────┬──────────────────┐    │
│ │  FU   │  FU   │                   │    │
│ │indicator│header│  分片数据 #N       │    │
│ └──────┴───────┴──────────────────┘    │
└─────────────────────────────────────────┘
```

每个 FU-A 包有两个字节的头：

**FU indicator（1 字节）**：
```
+---+---+---+---+---+---+---+---+
|F=0|NRI|    Type = 28 (FU-A)    |
+---+---+---+---+---+---+---+---+
```
- F（1 bit）：forbidden_zero_bit，固定 0
- NRI（2 bit）：优先级，从原始 NAL header 复制
- Type（5 bit）：固定 28，表示 FU-A

**FU header（1 字节）**：
```
+---+---+---+---+---+---+---+---+
|S=1|E=0|R=0|  Type (原始 NAL 类型) |
+---+---+---+---+---+---+---+---+
```
- S（1 bit）：Start，第一片为 1，其余为 0
- E（1 bit）：End，最后一片为 1，其余为 0
- R（1 bit）：保留位，固定 0
- Type（5 bit）：原始 NAL 单元的类型（如 5=IDR 帧）

所以一个 FU-A 包的实际负载 = MTU - IP - UDP - RTP 头 - 2 字节 FU 头 = 约 **1458 字节**。

# 打包端：把 H.264 帧拆成 FU-A 包

```c
#include <stdint.h>
#include <string.h>
#include <arpa/inet.h>  // htonl, htons

#define RTP_HEADER_SIZE    12
#define FU_HEADER_SIZE      2
#define MAX_RTP_PAYLOAD  1400   // 每片最大数据量

typedef struct {
    // RTP 固定头
    uint8_t  version;         // 2
    uint8_t  padding;         // 0
    uint8_t  extension;       // 0
    uint8_t  csrc_count;      // 0
    uint8_t  marker;          // 最后一包为 1
    uint8_t  payload_type;    // 96 = H.264
    uint16_t sequence_number;
    uint32_t timestamp;
    uint32_t ssrc;
    
    // 负载
    uint8_t  payload[MAX_RTP_PAYLOAD];
    int      payload_size;
} RtpPacket;

// 打包一个 RTP 包（写入 UDP socket）
int rtp_packet_send(int sock, RtpPacket *pkt, struct sockaddr_in *dest) {
    uint8_t buffer[1500];
    int offset = 0;
    
    // RTP 固定头（12 字节）
    buffer[0] = (pkt->version << 6) | (pkt->padding << 5) |
                (pkt->extension << 4) | pkt->csrc_count;
    buffer[1] = (pkt->marker << 7) | pkt->payload_type;
    *(uint16_t *)(buffer + 2) = htons(pkt->sequence_number);
    *(uint32_t *)(buffer + 4) = htonl(pkt->timestamp);
    *(uint32_t *)(buffer + 8) = htonl(pkt->ssrc);
    offset += RTP_HEADER_SIZE;
    
    // 负载
    memcpy(buffer + offset, pkt->payload, pkt->payload_size);
    offset += pkt->payload_size;
    
    return sendto(sock, buffer, offset, 0,
                  (struct sockaddr *)dest, sizeof(*dest));
}

// 将一个 H.264 NAL 单元分片打包为多个 FU-A RTP 包
int rtp_send_h264_fu_a(int sock, const uint8_t *nalu, int nalu_size,
                        uint32_t timestamp, struct sockaddr_in *dest) {
    if (nalu_size <= MAX_RTP_PAYLOAD) {
        // 不需要分片，单一 NAL 单元模式
        RtpPacket pkt = {
            .version = 2, .payload_type = 96, .marker = 1,
            .timestamp = timestamp, .payload_size = nalu_size
        };
        memcpy(pkt.payload, nalu, nalu_size);
        return rtp_packet_send(sock, &pkt, dest);
    }
    
    uint8_t nalu_header = nalu[0];           // 原始 NAL header
    int nalu_type = nalu_header & 0x1F;      // 低 5 位是类型
    int nri       = (nalu_header >> 5) & 0x03; // 第 6-7 位是 NRI
    
    const uint8_t *data = nalu + 1;  // 跳过 NAL header
    int data_size = nalu_size - 1;
    int offset = 0;
    int total_packets = (data_size + MAX_RTP_PAYLOAD - FU_HEADER_SIZE - 1)
                        / (MAX_RTP_PAYLOAD - FU_HEADER_SIZE);
    
    for (int i = 0; i < total_packets; i++) {
        RtpPacket pkt = {
            .version = 2, .payload_type = 96,
            .timestamp = timestamp,
            .marker = (i == total_packets - 1) ? 1 : 0  // 最后一包设 marker
        };
        pkt.sequence_number++;  // 每个包递增
        
        // 构建 FU indicator 和 FU header
        int fu_payload_offset = 0;
        
        // FU indicator = F(0) | NRI(2bit) | Type=28(5bit)
        pkt.payload[fu_payload_offset++] = (nri << 5) | 28;
        
        // FU header = S(1bit) | E(1bit) | R(1bit) | Type(5bit)
        uint8_t fu_header = nalu_type & 0x1F;
        if (i == 0)               fu_header |= 0x80;  // Start
        if (i == total_packets - 1) fu_header |= 0x40;  // End
        pkt.payload[fu_payload_offset++] = fu_header;
        
        // 拷贝分片数据
        int chunk_size = MAX_RTP_PAYLOAD - fu_payload_offset;
        if (offset + chunk_size > data_size)
            chunk_size = data_size - offset;
        
        memcpy(pkt.payload + fu_payload_offset, data + offset, chunk_size);
        pkt.payload_size = fu_payload_offset + chunk_size;
        offset += chunk_size;
        
        rtp_packet_send(sock, &pkt, dest);
    }
    
    return total_packets;
}
```

# 解包端：把 FU-A 包拼回完整帧

接收端的逻辑反过来：维护一个"待组装帧"的缓冲，FU-A Start 包重置缓冲，Middle 包追加数据，End 包完成组装。

```c
typedef struct {
    uint8_t  buffer[512 * 1024];    // 最多 512KB 一帧
    int      buffer_size;
    int      assembling;            // 是否正在组装
    uint16_t expected_seq;          // 期望的下一个序列号
    uint32_t ssrc;
    uint8_t  nalu_type;             // 原始 NAL 类型
} FuAssembler;

// 解码一个 FU-A 包，拼回完整帧
// 返回值：>0 = 完整帧大小，0 = 继续等待，-1 = 错误
int fu_assemble(FuAssembler *fa, const uint8_t *rtp_payload,
                int payload_size, uint16_t seq) {
    if (payload_size < 2) return -1;
    
    const uint8_t *fu_data = rtp_payload + FU_HEADER_SIZE;
    int fu_data_size = payload_size - FU_HEADER_SIZE;
    
    uint8_t fu_indicator = rtp_payload[0];
    uint8_t fu_header    = rtp_payload[1];
    
    int start = (fu_header >> 7) & 0x01;
    int end   = (fu_header >> 6) & 0x01;
    int nalu_type = fu_header & 0x1F;
    
    if (start) {
        // 开始新的一帧
        fa->assembling = 1;
        fa->buffer_size = 0;
        fa->nalu_type = nalu_type;
        
        // 写入重建的 NAL header
        uint8_t nri = (fu_indicator >> 5) & 0x03;
        fa->buffer[0] = (nri << 5) | (nalu_type & 0x1F);
        fa->buffer_size = 1;
    }
    
    if (!fa->assembling) return -1;
    
    // 检查序列号连续性（可选，检测丢包）
    if (fa->expected_seq != 0 && seq != fa->expected_seq) {
        fprintf(stderr, "RTP 包乱序或丢包: 期望 %d, 收到 %d\n",
                fa->expected_seq, seq);
        // 丢包严重时可以直接放弃当前帧
        if (abs((int)(seq - fa->expected_seq)) > 3) {
            fa->assembling = 0;
            return -1;
        }
    }
    fa->expected_seq = seq + 1;
    
    // 追加分片数据
    if (fa->buffer_size + fu_data_size > sizeof(fa->buffer)) {
        fprintf(stderr, "帧缓冲溢出\n");
        fa->assembling = 0;
        return -1;
    }
    
    memcpy(fa->buffer + fa->buffer_size, fu_data, fu_data_size);
    fa->buffer_size += fu_data_size;
    
    if (end) {
        // 最后一包，帧组装完成
        fa->assembling = 0;
        fa->expected_seq = 0;
        return fa->buffer_size;
    }
    
    return 0;  // 等待更多分片
}
```

# RTP 时间戳的正确理解

RTP 时间戳**不是绝对时间**，它是采样时刻的计数器。对于 H.264 视频，时钟频率固定为 **90kHz**（即每秒 90000 个 tick）。

```
RTP 时间戳的含义 = (timestamp / 90000) 秒
```

相邻两帧的 RTP 时间戳差值 = `90000 / 帧率`。比如 30fps：`90000 / 30 = 3000`。

```c
// 将 RTP 时间戳换算为毫秒
uint64_t rtp_ts_to_ms(uint32_t rtp_ts, uint32_t clock_rate) {
    return (uint64_t)rtp_ts * 1000 / clock_rate;
}

// 视频：90kHz 时钟
uint64_t pts_ms = rtp_ts_to_ms(pkt.timestamp, 90000);

// 音频：采样率时钟（如 44.1kHz）
uint64_t pts_ms = rtp_ts_to_ms(pkt.timestamp, 44100);
```

RTP 包头里还有一个 32 位 `timestamp` 字段——它会在 32 位溢出后回绕。处理回绕需要特殊逻辑：

```c
int32_t rtp_ts_diff(uint32_t ts_new, uint32_t ts_old) {
    // RTCP 规范定义的回绕安全差值
    return (int32_t)(ts_new - ts_old);  // 利用有符号溢出的特性
}
```

RTP 的初始时间戳是一个随机值（安全考虑）。所以不能直接用 `timestamp / clock_rate` 算"绝对播放时间"，而应该算**差值**：当前包的时间戳减去第一个包的时间戳。

# 完整接收端的骨架

```c
typedef struct {
    int sock;
    FuAssembler fu_asm;
    
    uint32_t ssrc;
    uint32_t base_timestamp;   // 第一个包的 timestamp
    uint64_t base_ntp;         // 第一个包到达的 NTP 时间
    int      base_set;
} RtpReceiver;

int rtp_receive_loop(RtpReceiver *rx) {
    uint8_t buf[2048];
    struct sockaddr_in src;
    socklen_t src_len = sizeof(src);
    
    while (running) {
        int len = recvfrom(rx->sock, buf, sizeof(buf), 0,
                           (struct sockaddr *)&src, &src_len);
        if (len < RTP_HEADER_SIZE) continue;
        
        // 解析 RTP 固定头
        uint8_t  version   = (buf[0] >> 6) & 0x03;
        uint16_t seq       = ntohs(*(uint16_t *)(buf + 2));
        uint32_t timestamp = ntohl(*(uint32_t *)(buf + 4));
        uint32_t ssrc      = ntohl(*(uint32_t *)(buf + 8));
        uint8_t  pt        = buf[1] & 0x7F;
        
        // 记录基准时间戳（用于后续 PTS 换算）
        if (!rx->base_set) {
            rx->ssrc = ssrc;
            rx->base_timestamp = timestamp;
            rx->base_ntp = av_gettime_relative();  // 微秒
            rx->base_set = 1;
        }
        
        if (ssrc != rx->ssrc) continue;  // 不是我们的流
        
        // RTP 负载
        uint8_t *payload = buf + RTP_HEADER_SIZE;
        int payload_size = len - RTP_HEADER_SIZE;
        
        // FU-A 解包
        int frame_size = fu_assemble(&rx->fu_asm, payload, payload_size, seq);
        if (frame_size > 0) {
            // 拿到完整帧，交给解码器
            uint32_t ts_diff = timestamp - rx->base_timestamp;
            int64_t pts_us = rx->base_ntp + rtp_ts_to_ms(ts_diff, 90000) * 1000;
            on_frame_complete(rx->fu_asm.buffer, frame_size, pts_us);
        }
    }
    
    return 0;
}
```

# 小结

手写一遍 RTP FU-A 打包和解包，你会对这几个点有质变级的理解：

| 概念 | 关键点 |
|------|--------|
| FU-A 分片 | 2 字节头（indicator + header），S/E 标志位标记首尾 |
| RTP 时间戳 | 不是绝对时间，是采样计数，视频 90kHz 时钟 |
| 序列号 | 16 位，检测丢包和乱序，解包时必须处理回绕 |
| Marker 位 | 最后一包置 1，帮助接收端判断帧边界 |
| 包重组 | Start 重置缓冲 → Middle 追加 → End 完成 |

虽然 FFmpeg 已经实现了所有这些，但自己写过一遍之后，再看 `av_read_frame` 返回的包就知道它背后做了什么——RTP 收包、重排序、Jitter Buffer、FU-A 重组。下一篇用一个更实际的需求串联这些知识：多路流的混音和画面拼接。
